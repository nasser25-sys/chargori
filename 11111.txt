    <script>
        let cart = [];

        document.addEventListener('DOMContentLoaded', () => {
            const savedCart = localStorage.getItem('chargoryCart');
            if (savedCart) {
                cart = JSON.parse(savedCart);
                updateCartDisplay();
            }
            // Removed initThreeJs() call
        });

        // --- Cart Functions ---
        function saveCart() {
            localStorage.setItem('chargoryCart', JSON.stringify(cart));
        }

        function updateCartDisplay() {
            const cartItemCount = document.getElementById('cartItemCount');
            const totalItemsInCart = cart.reduce((sum, item) => sum + item.quantity, 0);
            cartItemCount.innerText = totalItemsInCart;
            cartItemCount.style.display = totalItemsInCart > 0 ? 'block' : 'none';
        }

        function addToCart(productName, productPrice) {
            const existingItemIndex = cart.findIndex(item => item.name === productName);
            if (existingItemIndex > -1) {
                cart[existingItemIndex].quantity += 1;
            } else {
                cart.push({ name: productName, price: productPrice, quantity: 1 });
            }
            saveCart();
            updateCartDisplay();
            alert(`تم إضافة "${productName}" إلى سلة التسوق!`);
        }

        function openCartModal() {
            const cartItemsList = document.getElementById('cartItemsList');
            const cartTotalElement = document.getElementById('cartTotal');
            const emptyCartMessage = document.getElementById('emptyCartMessage');
            const proceedToCheckoutBtn = document.getElementById('proceedToCheckoutBtn');

            cartItemsList.innerHTML = '';

            if (cart.length === 0) {
                emptyCartMessage.style.display = 'block';
                proceedToCheckoutBtn.disabled = true;
            } else {
                emptyCartMessage.style.display = 'none';
                let total = 0;
                cart.forEach((item, index) => {
                    const itemElement = document.createElement('div');
                    itemElement.classList.add('cart-item');
                    itemElement.innerHTML = `
                        <button class="qty-button text-red-600" onclick="removeFromCart(${index})">X</button>
                        <div class="cart-item-info">
                            <h4 class="font-semibold">${item.name}</h4>
                            <p>${item.price.toFixed(2)} درهم مغربي</p>
                        </div>
                        <div class="cart-item-quantity-controls">
                            <button class="qty-button" onclick="updateQuantity(${index}, ${item.quantity - 1})">-</button>
                            <span class="mx-2 font-bold">${item.quantity}</span>
                            <button class="qty-button" onclick="updateQuantity(${index}, ${item.quantity + 1})">+</button>
                        </div>
                    `;
                    cartItemsList.appendChild(itemElement);
                    total += item.price * item.quantity;
                });
                cartTotalElement.innerText = total.toFixed(2);
                proceedToCheckoutBtn.disabled = false;
            }

            document.getElementById('cartModal').classList.add('active');
        }

        function closeCartModal() {
            document.getElementById('cartModal').classList.remove('active');
        }

        function removeFromCart(index) {
            cart.splice(index, 1);
            saveCart();
            updateCartDisplay();
            openCartModal();
        }

        function updateQuantity(index, newQuantity) {
            if (newQuantity < 1) {
                removeFromCart(index);
            } else {
                cart[index].quantity = newQuantity;
                saveCart();
                updateCartModal();
            }
        }

        function proceedToCheckout() {
            closeCartModal();
            const totalCartAmount = cart.reduce((sum, item) => sum + item.price * item.quantity, 0);
            document.getElementById('checkoutTotalAmount').innerText = totalCartAmount.toFixed(2);

            document.getElementById('onlinePaymentOption').classList.remove('selected');
            document.getElementById('codPaymentOption').classList.remove('selected');
            document.getElementById('paymentDetails').classList.add('hidden');
            document.getElementById('onlinePaymentForm').classList.add('hidden');
            document.getElementById('codConfirmation').classList.add('hidden');
            document.getElementById('confirmationMessage').classList.add('hidden');
            document.getElementById('confirmationMessage').innerText = '';
            selectedPaymentMethod = '';

            document.getElementById('paymentModal').classList.add('active');
        }

        // --- Payment Functions ---
        let selectedPaymentMethod = '';

        function closePaymentOptions() {
            document.getElementById('paymentModal').classList.remove('active');
        }

        function selectPaymentMethod(method) {
            selectedPaymentMethod = method;

            document.getElementById('onlinePaymentOption').classList.remove('selected');
            document.getElementById('codPaymentOption').classList.remove('selected');

            document.getElementById('onlinePaymentForm').classList.add('hidden');
            document.getElementById('codConfirmation').classList.add('hidden');
            document.getElementById('confirmationMessage').classList.add('hidden');

            document.getElementById('paymentDetails').classList.remove('hidden');

            if (method === 'online') {
                document.getElementById('onlinePaymentOption').classList.add('selected');
                document.getElementById('paymentDetailsTitle').innerText = 'أدخل تفاصيل البطاقة';
                document.getElementById('onlinePaymentForm').classList.remove('hidden');
            } else if (method === 'cod') {
                document.getElementById('codPaymentOption').classList.add('selected');
                document.getElementById('paymentDetailsTitle').innerText = 'معلومات التوصيل';
                document.getElementById('codConfirmation').classList.remove('hidden');
            }
        }

        function processPayment(method) {
            const confirmationMessageElement = document.getElementById('confirmationMessage');
            confirmationMessageElement.classList.remove('hidden');

            if (method === 'online') {
                confirmationMessageElement.innerText = `تم استلام طلبك بنجاح! سيتم معالجة الدفع عبر الإنترنت. (إجمالي: ${document.getElementById('checkoutTotalAmount').innerText} درهم)`;
                confirmationMessageElement.classList.remove('text-red-600');
                confirmationMessageElement.classList.add('text-green-600');
            } else if (method === 'cod') {
                confirmationMessageElement.innerText = `تم تأكيد طلبك بنجاح! سيتم الدفع عند الاستلام. (إجمالي: ${document.getElementById('checkoutTotalAmount').innerText} درهم)`;
                confirmationMessageElement.classList.remove('text-red-600');
                confirmationMessageElement.classList.add('text-green-600');
            }

            cart = [];
            saveCart();
            updateCartDisplay();

            setTimeout(() => {
                closePaymentOptions();
            }, 3000);
        }
